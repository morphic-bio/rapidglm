cmake_minimum_required(VERSION 3.16)
project(rapidglm VERSION 1.4.1 LANGUAGES C CXX)
option(RAPIDGLM_OPENMP "Enable OpenMP" ON)
option(RAPIDGLM_FORCE_PORTABLE "Force portable kernels and integer/SHA paths on x86" OFF)
option(RAPIDGLM_TESTS "Build tests" ON)
option(RAPIDGLM_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(RAPIDGLM_BENCHMARKS "Build isolated numerical benchmarks" OFF)
option(RAPIDGLM_SOLVER_BENCHMARK "Build the solver comparison against Ceres Solver (needs Ceres)" OFF)
option(RAPIDGLM_PYTHON "Build the Python extension module (needs pybind11)" OFF)
option(RAPIDGLM_LAPACK_SOLVES "Use LAPACK for Cholesky factorizations above 16 columns and for QR triangular solves (see docs/CLI.md)" OFF)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(Boost 1.65 REQUIRED)
find_package(LAPACK REQUIRED)
find_package(LAPACKE REQUIRED)
add_library(rapidglm INTERFACE)
add_library(rapidglm::rapidglm ALIAS rapidglm)
target_compile_features(rapidglm INTERFACE cxx_std_17)
if(RAPIDGLM_FORCE_PORTABLE)
  target_compile_definitions(rapidglm INTERFACE RAPIDGLM_FORCE_PORTABLE)
endif()
target_include_directories(rapidglm INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(rapidglm INTERFACE Boost::boost LAPACKE::LAPACKE)
if(RAPIDGLM_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(rapidglm INTERFACE OpenMP::OpenMP_CXX)
endif()
if(RAPIDGLM_LAPACK_SOLVES)
  target_compile_definitions(rapidglm INTERFACE RAPIDGLM_LAPACK_SOLVES)
  message(WARNING "RAPIDGLM_LAPACK_SOLVES is ON: Cholesky factorizations above 16 columns and QR "
                  "triangular solves use the linked LAPACK/BLAS. Those results then depend on that "
                  "library's kernels for this processor; you are responsible for checking them. "
                  "Run ctest on the machines that will run the fits (see docs/CLI.md).")
endif()
if(RAPIDGLM_SANITIZERS)
  target_compile_options(rapidglm INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rapidglm INTERFACE -fsanitize=address,undefined)
endif()
# C interface (include/rapidglm/rapidglm_c.h) for bindings in other languages.
add_library(rapidglm_c SHARED src/c_api.cpp)
add_library(rapidglm_c_static STATIC src/c_api.cpp)
foreach(target rapidglm_c rapidglm_c_static)
  target_link_libraries(${target} PRIVATE rapidglm)
  target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
set_target_properties(rapidglm_c PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR})
# Keep the directories of a BLAS/LAPACK outside the system paths (such as a
# local OpenBLAS) in the installed library's runpath.
set(_rapidglm_c_rpath "")
foreach(_lib IN LISTS LAPACK_LIBRARIES)
  if(IS_ABSOLUTE "${_lib}")
    get_filename_component(_dir "${_lib}" DIRECTORY)
    if(NOT _dir IN_LIST CMAKE_CXX_IMPLICIT_LINK_DIRECTORIES)
      list(APPEND _rapidglm_c_rpath "${_dir}")
    endif()
  endif()
endforeach()
list(REMOVE_DUPLICATES _rapidglm_c_rpath)
set_target_properties(rapidglm_c PROPERTIES INSTALL_RPATH "${_rapidglm_c_rpath}")
set_target_properties(rapidglm_c_static PROPERTIES OUTPUT_NAME rapidglm_c)
add_executable(rapidglm_cli src/main.cpp)
set_target_properties(rapidglm_cli PROPERTIES OUTPUT_NAME rapidglm)
target_link_libraries(rapidglm_cli PRIVATE rapidglm)
target_compile_options(rapidglm_cli PRIVATE -Wall -Wextra -Wpedantic)
if(RAPIDGLM_TESTS)
  enable_testing()
  add_executable(rapidglm_tests tests/test_engine.cpp tests/test_numerics.cpp tests/test_generator.cpp tests/test_boundary.cpp tests/test_effects.cpp tests/test_sceptre_model.cpp tests/test_storage.cpp tests/test_solver.cpp tests/test_quasi.cpp tests/test_scan.cpp tests/test_bulk.cpp tests/test_hypothesis.cpp tests/test_c_api.cpp tests/test_batching.cpp tests/test_counts.cpp tests/test_qtl.cpp tests/test_edger.cpp)
  target_link_libraries(rapidglm_tests PRIVATE rapidglm rapidglm_c_static)
  add_test(NAME engine COMMAND rapidglm_tests)
  add_executable(rapidglm_c_test tests/test_c_api.c)
  target_link_libraries(rapidglm_c_test PRIVATE rapidglm_c m)
  set_target_properties(rapidglm_c_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
  target_compile_options(rapidglm_c_test PRIVATE -Wall -Wextra -Wpedantic)
  if(RAPIDGLM_SANITIZERS)
    target_compile_options(rapidglm_c_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(rapidglm_c_test PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME c_api COMMAND rapidglm_c_test)
  set_tests_properties(c_api PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1")
  # Limit internal workers while testing concurrent calls from outer OpenMP.
  # This still requires a thread-safe BLAS build (see README).
  set_tests_properties(engine PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS rapidglm rapidglm_c rapidglm_c_static rapidglm_cli EXPORT rapidglmTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT rapidglmTargets NAMESPACE rapidglm:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rapidglm)
configure_package_config_file(cmake/rapidglmConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/rapidglmConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rapidglm)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/rapidglmConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/rapidglmConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/rapidglmConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rapidglm)
install(FILES cmake/FindLAPACKE.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rapidglm)
if(RAPIDGLM_SOLVER_BENCHMARK)
  find_package(Ceres REQUIRED)
  add_executable(rapidglm_solver_benchmark benchmarks/solvers/compare_solvers.cpp)
  target_include_directories(rapidglm_solver_benchmark PRIVATE src)
  target_link_libraries(rapidglm_solver_benchmark PRIVATE rapidglm Ceres::ceres)
endif()
if(RAPIDGLM_BENCHMARKS)
  add_executable(rapidglm_benchmarks benchmarks/kernels.cpp)
  target_link_libraries(rapidglm_benchmarks PRIVATE rapidglm)
  add_executable(rapidglm_effect_benchmark benchmarks/effects.cpp)
  target_link_libraries(rapidglm_effect_benchmark PRIVATE rapidglm)
  add_executable(rapidglm_mask_benchmark benchmarks/masks.cpp)
  target_link_libraries(rapidglm_mask_benchmark PRIVATE rapidglm)
  add_executable(rapidglm_glm_compare benchmarks/glm/rapidglm_fit.cpp)
  target_link_libraries(rapidglm_glm_compare PRIVATE rapidglm)
  add_executable(rapidglm_glm_batch benchmarks/glm/rapidglm_batch.cpp)
  target_link_libraries(rapidglm_glm_batch PRIVATE rapidglm)
endif()
if(RAPIDGLM_TESTS)
  add_executable(rapidglm_calibration tests/calibration.cpp)
  target_link_libraries(rapidglm_calibration PRIVATE rapidglm)
  add_executable(rapidglm_boundary_diagnostic tests/diagnose_boundary.cpp)
  target_link_libraries(rapidglm_boundary_diagnostic PRIVATE rapidglm)
  add_executable(rapidglm_calibration_audit tests/audit_calibration.cpp)
  target_link_libraries(rapidglm_calibration_audit PRIVATE rapidglm)
  add_executable(rapidglm_calibration_sparse tests/calibration_sparse.cpp)
  target_link_libraries(rapidglm_calibration_sparse PRIVATE rapidglm)
endif()
if(RAPIDGLM_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_rapidglm bindings/python/src/module.cpp)
  target_link_libraries(_rapidglm PRIVATE rapidglm)
  set_target_properties(_rapidglm PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/bindings/python/rapidglm)
endif()
